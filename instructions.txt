# 1 - To handle cretificate issue, run following certificate with elevated permission:
# Define the DNS name and PFX file path
$dnsName = "YourDomainName"
$pfxPath = "D:\bakar.pfx"
$password = "bakar"

# Create the self-signed certificate
$cert = New-SelfSignedCertificate -DnsName $dnsName -CertStoreLocation "cert:\LocalMachine\My" -NotAfter (Get-Date).AddYears(10)

# Get the thumbprint of the created certificate
$thumbprint = $cert.Thumbprint

# Export the certificate to a PFX file using CertUtil
Start-Process -NoNewWindow -FilePath "certutil.exe" -ArgumentList "-exportPFX", "-p", $password, $thumbprint, $pfxPath -Wait
